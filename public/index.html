// 1. Conexión automática (funciona en PC, Móvil, Red local y Render) const socket = io(window.location.origin, { transports: ['websocket', 'polling'] }); // 2. Elementos de la pantalla const estadoConexion = document.getElementById('estado-conexion'); const mesaActualTexto = document.getElementById('mesa-actual'); const inputNota = document.getElementById('nota-pedido'); const btnEnviar = document.getElementById('btn-enviar'); const listaComandas = document.getElementById('lista-comandas'); let mesaSeleccionada = null; let productosPedido = []; // Identificamos quién somos (ej. /?rol=terraza) const urlParams = new URLSearchParams(window.location.search); const rol = urlParams.get('rol') || 'PC'; // 3. Gestión de la conexión socket.on('connect', () => { estadoConexion.textContent = `🟢 Conectado (${rol.toUpperCase()})`; estadoConexion.style.color = '#a8ffca'; }); socket.on('disconnect', () => { estadoConexion.textContent = `🔴 Desconectado`; estadoConexion.style.color = '#ff9999'; }); // 4. Lógica de selección de mesas document.querySelectorAll('.mesa-btn').forEach(btn => { btn.addEventListener('click', (e) => { // Quitar color activo de todas document.querySelectorAll('.mesa-btn').forEach(b => b.classList.remove('activa')); // Poner color activo a la pulsada e.target.classList.add('activa'); mesaSeleccionada = e.target.textContent; mesaActualTexto.textContent = mesaSeleccionada; }); }); // 5. Lógica de añadir productos rápidos document.querySelectorAll('.prod-btn').forEach(btn => { btn.addEventListener('click', (e) => { const producto = e.target.textContent; productosPedido.push(producto); inputNota.value += (inputNota.value ? ', ' : '') + producto; }); }); // 6. Enviar la comanda al servidor btnEnviar.addEventListener('click', () => { if (!mesaSeleccionada) { alert('⚠️ Por favor, selecciona una mesa primero'); return; } if (!inputNota.value.trim()) { alert('⚠️ El pedido está vacío'); return; } const comanda = { mesa: mesaSeleccionada, pedido: inputNota.value, hora: new Date().toLocaleTimeString(), origen: rol.toUpperCase() }; socket.emit('nueva-comanda', comanda); // Limpiar para el siguiente pedido inputNota.value = ''; productosPedido = []; document.querySelectorAll('.mesa-btn').forEach(b => b.classList.remove('activa')); mesaSeleccionada = null; mesaActualTexto.textContent = 'Ninguna'; }); // 7. Recibir comandas de la cocina / servidor socket.on('actualizar-comandas', (comanda) => { const li = document.createElement('li'); li.className = 'comanda-item'; li.innerHTML = ` ⏰ ${comanda.hora} | 📍 Mesa: ${comanda.mesa} | 👤 ${comanda.origen} ${comanda.pedido} `; // Poner arriba del todo listaComandas.prepend(li); });
